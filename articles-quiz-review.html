import React, { useState } from 'react';

import { CheckCircle, XCircle, Trophy, RefreshCw, User, BookOpen } from 'lucide-react';


const QuizReview = () => {

  const paragraph = "Last week, I downloaded ___ new app that helps students manage their time better. ___ app is called StudyBuddy, and it was recommended by ___ friend in my psychology class. Like many students, I waste ___ hour or two every day scrolling through ___ social media. ___ Instagram and TikTok are ___ biggest distractions for me. According to ___ data it collected, I spend about three hours per day on ___ entertainment apps! Now I'm trying to make ___ change by turning off ___ notifications on my phone. If you're ___ student who struggles with ___ time management, I definitely recommend trying this app.";


  const answers = ["a", "the", "a", "an", "no article", "no article", "the", "the", "no article", "a", "the", "a", "no article"];

  

  const explanations = [

    "Use 'a' before 'new' (consonant sound) when mentioning the app for the first time.",

    "Use 'the' for the specific app mentioned in the previous sentence.",

    "Use 'a' before 'friend' (consonant sound) - one friend among many.",

    "Use 'an' before 'hour' (silent h makes a vowel sound).",

    "No article before 'social media' - it's a general concept.",

    "No article before 'Instagram' - it's a proper name/brand.",

    "Use 'the' before the superlative 'biggest'.",

    "Use 'the' for the specific data mentioned (from the app).",

    "No article before 'entertainment' when used as a general adjective describing types of apps.",

    "Use 'a' before 'change' (consonant sound) - making one change.",

    "Use 'the' for the specific notifications on my phone.",

    "Use 'a' before 'student' (consonant sound) - any student.",

    "No article before 'time management' - it's a general concept/skill."

  ];


  const [screen, setScreen] = useState('intro');

  const [studentName, setStudentName] = useState('');

  const [currentBlank, setCurrentBlank] = useState(0);

  const [userAnswers, setUserAnswers] = useState(Array(13).fill(null));

  const [showResults, setShowResults] = useState(false);


  const startReview = () => {

    if (studentName.trim()) {

      setScreen('review');

    }

  };


  const handleAnswer = (option) => {

    const newAnswers = [...userAnswers];

    newAnswers[currentBlank] = option;

    setUserAnswers(newAnswers);

    

    if (currentBlank < 12) {

      setCurrentBlank(currentBlank + 1);

    } else {

      setShowResults(true);

    }

  };


  const resetReview = () => {

    setScreen('intro');

    setStudentName('');

    setCurrentBlank(0);

    setUserAnswers(Array(13).fill(null));

    setShowResults(false);

  };


  const renderParagraph = () => {

    const parts = paragraph.split('___');

    return parts.map((part, idx) => (

      <span key={idx}>

        {part}

        {idx < parts.length - 1 && (

          <span className={`font-bold px-2 py-1 rounded ${

            idx === currentBlank && !showResults

              ? 'bg-gray-300'

              : userAnswers[idx]

              ? userAnswers[idx] === answers[idx]

                ? 'text-green-600'

                : 'text-red-600'

              : ''

          }`}>

            {showResults || idx < currentBlank

              ? userAnswers[idx] === 'no article' ? 'X' : userAnswers[idx]

              : '___'}

          </span>

        )}

      </span>

    ));

  };


  const score = userAnswers.filter((ans, idx) => ans === answers[idx]).length;


  // Introduction screen

  if (screen === 'intro') {

    return (

      <div className="min-h-screen bg-gradient-to-br from-gray-100 to-slate-200 flex items-center justify-center p-4">

        <div className="bg-white rounded-2xl shadow-2xl p-8 max-w-2xl w-full">

          <div className="text-center mb-8">

            <div className="bg-gray-200 w-20 h-20 rounded-full flex items-center justify-center mx-auto mb-4">

              <BookOpen className="w-10 h-10 text-gray-700" />

            </div>

            <h1 className="text-4xl font-bold text-gray-800 mb-2">Articles Quiz Review</h1>

            <h2 className="text-xl font-semibold text-gray-600">Practice Before Your In-Person Quiz</h2>

          </div>

          

          <div className="bg-blue-50 border-2 border-blue-300 rounded-lg p-6 mb-6">

            <h3 className="font-bold text-lg text-gray-800 mb-3">📋 Instructions:</h3>

            <ul className="space-y-2 text-gray-700">

              <li className="flex items-start gap-2">

                <span className="font-bold min-w-fit">•</span>

                <span>This is a <strong>completion grade</strong> - you get credit for doing it, not for your score</span>

              </li>

              <li className="flex items-start gap-2">

                <span className="font-bold min-w-fit">•</span>

                <span>Read the paragraph and fill in each blank with the correct article</span>

              </li>

              <li className="flex items-start gap-2">

                <span className="font-bold min-w-fit">•</span>

                <span>You'll get instant feedback with explanations</span>

              </li>

              <li className="flex items-start gap-2">

                <span className="font-bold min-w-fit">•</span>

                <span>Take a <strong>screenshot</strong> of your final result</span>

              </li>

              <li className="flex items-start gap-2">

                <span className="font-bold min-w-fit">•</span>

                <span>Submit it to <strong>Professor Bishop</strong> by Sunday night</span>

              </li>

            </ul>

          </div>


          <div className="mb-6">

            <label className="block text-sm font-semibold text-gray-700 mb-2">

              Enter your name:

            </label>

            <input

              type="text"

              value={studentName}

              onChange={(e) => setStudentName(e.target.value)}

              onKeyPress={(e) => e.key === 'Enter' && startReview()}

              placeholder="Your full name"

              className="w-full px-4 py-3 border-2 border-gray-300 rounded-lg focus:border-gray-500 focus:outline-none text-lg"

              autoFocus

            />

          </div>

          

          <button

            onClick={startReview}

            disabled={!studentName.trim()}

            className={`w-full py-3 rounded-lg font-semibold text-lg transition ${

              studentName.trim()

                ? 'bg-gray-700 text-white hover:bg-gray-800'

                : 'bg-gray-300 text-gray-500 cursor-not-allowed'

            }`}

          >

            Start Review

          </button>

        </div>

      </div>

    );

  }


  // Results screen

  if (showResults) {

    const percentage = Math.round((score / 13) * 100);

    const currentDate = new Date().toLocaleDateString();

    

    return (

      <div className="min-h-screen bg-gradient-to-br from-gray-100 to-slate-200 flex items-center justify-center p-4">

        <div className="bg-white rounded-2xl shadow-2xl p-8 max-w-3xl w-full">

          <div className="text-center mb-6">

            <Trophy className="w-16 h-16 mx-auto mb-4 text-gray-600" />

            <h2 className="text-3xl font-bold text-gray-800 mb-2">Review Complete!</h2>

            <p className="text-gray-600">Review your answers and explanations below</p>

          </div>

          

          <div className="bg-gray-100 rounded-lg p-6 mb-6">

            <div className="text-lg font-semibold text-gray-700 mb-1">Student Name:</div>

            <div className="text-2xl font-bold text-gray-700 mb-4">{studentName}</div>

            

            <div className="text-lg font-semibold text-gray-700 mb-1">Your Score:</div>

            <p className="text-5xl font-bold text-gray-700 mb-2">{score}/13</p>

            <p className="text-xl text-gray-600 mb-4">{percentage}%</p>

            

            <div className="text-base bg-blue-100 text-blue-800 p-3 rounded-lg font-semibold">

              ✓ Completion Credit Earned

            </div>

            

            <div className="text-sm text-gray-500 mt-4">{currentDate}</div>

          </div>


          <div className="bg-yellow-50 border-2 border-yellow-300 rounded-lg p-4 mb-6">

            <p className="font-bold text-gray-800 mb-2">📸 IMPORTANT:</p>

            <p className="text-sm text-gray-700">

              Take a screenshot of this page and submit it to <strong>Professor Bishop</strong> by Sunday night for completion credit.

            </p>

          </div>


          <div className="bg-white border-2 border-gray-300 rounded-lg p-6 mb-6">

            <h3 className="font-bold text-xl text-gray-800 mb-4">📖 Paragraph with Your Answers:</h3>

            <p className="text-lg text-gray-700 leading-relaxed mb-6">

              {renderParagraph()}

            </p>

          </div>


          <div className="space-y-4 mb-6">

            <h3 className="font-bold text-xl text-gray-800">📝 Answer Review:</h3>

            {answers.map((correctAnswer, idx) => (

              <div key={idx} className={`p-4 rounded-lg border-2 ${

                userAnswers[idx] === correctAnswer

                  ? 'bg-green-50 border-green-300'

                  : 'bg-red-50 border-red-300'

              }`}>

                <div className="flex items-start gap-3 mb-2">

                  {userAnswers[idx] === correctAnswer ? (

                    <CheckCircle className="w-6 h-6 text-green-600 flex-shrink-0 mt-1" />

                  ) : (

                    <XCircle className="w-6 h-6 text-red-600 flex-shrink-0 mt-1" />

                  )}

                  <div className="flex-1">

                    <div className="font-semibold text-gray-800 mb-1">

                      Blank #{idx + 1}

                    </div>

                    <div className="text-sm">

                      <span className="font-semibold">Your answer: </span>

                      <span className={userAnswers[idx] === correctAnswer ? 'text-green-700' : 'text-red-700'}>

                        {userAnswers[idx]}

                      </span>

                      {userAnswers[idx] !== correctAnswer && (

                        <span className="text-gray-700">

                          {' '} → Correct answer: <strong>{correctAnswer}</strong>

                        </span>

                      )}

                    </div>

                    <div className="text-sm text-gray-700 mt-2">

                      <strong>Explanation:</strong> {explanations[idx]}

                    </div>

                  </div>

                </div>

              </div>

            ))}

          </div>

          

          <button

            onClick={resetReview}

            className="bg-gray-700 text-white px-8 py-3 rounded-lg font-semibold hover:bg-gray-800 transition flex items-center gap-2 mx-auto"

          >

            <RefreshCw className="w-5 h-5" />

            Try Again

          </button>

        </div>

      </div>

    );

  }


  // Review screen

  return (

    <div className="min-h-screen bg-gradient-to-br from-gray-100 to-slate-200 flex items-center justify-center p-4">

      <div className="bg-white rounded-2xl shadow-2xl p-8 max-w-3xl w-full">

        <div className="flex justify-between items-center mb-6">

          <div className="flex items-center gap-2">

            <User className="w-5 h-5 text-gray-700" />

            <span className="font-semibold text-gray-700">{studentName}</span>

          </div>

          <div className="text-sm font-semibold text-gray-600">

            Blank {currentBlank + 1} / 13

          </div>

          <div className="flex items-center gap-2 bg-gray-200 px-4 py-2 rounded-full">

            <span className="font-bold text-gray-700">{currentBlank}/13</span>

          </div>

        </div>


        <div className="mb-8">

          <div className="w-full bg-gray-200 rounded-full h-2 mb-6">

            <div

              className="bg-gray-700 h-2 rounded-full transition-all duration-300"

              style={{ width: `${((currentBlank + 1) / 13) * 100}%` }}

            />

          </div>

          

          <h3 className="text-2xl font-bold text-gray-800 mb-4">

            Fill in blank #{currentBlank + 1}:

          </h3>

          <div className="bg-gray-50 p-6 rounded-lg border-2 border-gray-300 mb-6">

            <p className="text-xl text-gray-700 leading-relaxed">

              {renderParagraph()}

            </p>

          </div>

        </div>


        <div className="grid grid-cols-2 gap-4">

          {["a", "an", "the", "no article"].map((option, idx) => (

            <button

              key={idx}

              onClick={() => handleAnswer(option)}

              className="p-4 rounded-xl font-semibold text-lg transition-all bg-gray-100 hover:bg-gray-200 text-gray-800 hover:scale-105"

            >

              {option}

            </button>

          ))}

        </div>

      </div>

    </div>

  );

};


export default QuizReview;
